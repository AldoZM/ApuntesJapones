\subsection*{\emoji{speech-balloon} Frases Clave}
\frase{一つだけ\ruby{食}{た}べます。}{Solo comeré uno. (neutro)}
\frase{100\ruby{円}{えん}しかありません。}{Solo tengo 100 yenes. (insuficiente)}
\frase{ゲームばかりしている。}{No hace más que jugar videojuegos. (exceso)}
\frase{\ruby{勉強}{べんきょう}するだけでいい。}{Solo hay que estudiar.}

\subsection*{\emoji{gear} Gramática}
\patron{だけ — solo, únicamente (neutro)}
  {[N / V-dict] + だけ}
  {\ej{一\ruby{枚}{まい}だけください。}{Solo uno, por favor.}
   \ej{\ruby{見}{み}るだけでいいです。}{Solo mirarlo está bien.}
   \ej{できるだけ\ruby{早}{はや}く\ruby{来}{き}てください。}{Venga lo antes posible.}}

\patron{しか〜ない — solo (con connotación de insuficiencia)}
  {[N] + しか + negativo}
  {\ej{100\ruby{円}{えん}しかありません。}{Solo tengo 100 yenes. (no es suficiente)}
   \ej{\ruby{彼女}{かのじょ}しか\ruby{知}{し}らない。}{Solo ella lo sabe. (nadie más)}
   \ej{あと5\ruby{分}{ふん}しかない。}{Solo quedan 5 minutos.}}

\patron{ばかり — no hace más que / lleno de exceso}
  {[N / V-te] + ばかり}
  {\ej{ゲームばかりしている。}{No hace más que jugar videojuegos.}
   \ej{\ruby{甘}{あま}いものばかり\ruby{食}{た}べる。}{Solo come cosas dulces (preocupante).}
   \ej{さっき\ruby{来}{き}たばかりです。}{Acaba de llegar. (hace apenas un momento)}}

\comparacion{だけ vs しか vs ばかり}
  {だけ (neutro)\\\small limitación sin juicio\\\small \ruby{一}{ひと}つだけ\\\textit{Solo uno (está bien)}}
  {しか〜ない (insuficiencia)\\\small + ばかり (exceso crítico)\\\small 100\ruby{円}{えん}しかない\\\textit{Solo 100¥ (no alcanza)}}
  {だけ es neutro. しか implica que la cantidad es insuficiente. ばかり implica que se hace demasiado de algo, generalmente con tono crítico o humorístico.}

\coloquial{ばかり → ばっかり / ばっか (muy casual)}{
  En habla informal: ばかり → ばっかり → ばっか. «ゲームばっかじゃん！» (¡Solo haces videojuegos!) Es muy frecuente en la lengua de los jóvenes.}

\culturabox{\ruby{節制}{せっせい}と\ruby{過剰}{かじょう} — Moderación vs exceso}{
  En la cultura japonesa, el exceso (\ruby{過剰}{かじょう}) es mal visto socialmente. ばかり lleva una connotación crítica porque implica falta de equilibrio o moderación (\ruby{節制}{せっせい}). Decirle a alguien 「ゲームばかりしている」implica una suave reprimenda social.
}

\lectura{\ruby{健康診断}{けんこうしんだん}\\[0.4em]
  「\ruby{先生}{せんせい}、\ruby{最近}{さいきん}\ruby{体調}{たいちょう}が\ruby{悪}{わる}くて。」「\ruby{野菜}{やさい}はどれくらい\ruby{食}{た}べていますか？」「\ruby{肉}{にく}ばかり\ruby{食}{た}べています。\ruby{野菜}{やさい}はほとんど\ruby{食}{た}べません。」「\ruby{肉}{にく}だけじゃなくて、\ruby{野菜}{やさい}も\ruby{食}{た}べてください。」}
{Examen médico\\[0.4em]
  «Doctor, últimamente me siento mal.» «¿Cuántas verduras come?» «Solo como carne. Casi no como verduras.» «No solo carne, por favor coma verduras también.»}

\begin{tcolorbox}[ejerciciobox, title={\emoji{pencil} Ejercicios}]
\begin{enumerate}
  \item ¿だけ, しか, o ばかり? Solo tengo 500 yenes y no me alcanza.
  \item ¿だけ, しか, o ばかり? Mi hermano no hace más que dormir.
  \item Transforma usando ばかり: 彼は甘いものを食べる。
\end{enumerate}
\textbf{Respuestas:}
1) しか (500\ruby{円}{えん}しかない) \quad
2) ばかり (\ruby{弟}{おとうと}は\ruby{寝}{ね}てばかりいる) \quad
3) \ruby{彼}{かれ}は\ruby{甘}{あま}いものばかり\ruby{食}{た}べる
\end{tcolorbox}
