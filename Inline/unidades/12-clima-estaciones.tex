\subsection*{🗣️ Frases Clave}

\frase{\ruby{今日}{きょう}の\ruby{天気}{てんき}はどうですか？}{¿Cómo está el clima hoy?}
\frase{いい\ruby{天気}{てんき}ですね。}{Es un buen clima, ¿verdad?}
\frase{\ruby{明日}{あした}は\ruby{雨}{あめ}が\ruby{降}{ふ}るでしょう。}{Mañana lloverá (probablemente).}
\frase{\ruby{夏}{なつ}はとても\ruby{暑}{あつ}いです。}{El verano es muy caluroso.}

\subsection*{⚙️ Gramática}

\patron{Comparaciones simples}
  {A は B より Adj です (A es más adj que B)}
  {\ej{\ruby{夏}{なつ}は\ruby{春}{はる}より\ruby{暑}{あつ}いです。}{El verano es más caluroso que la primavera.}
  \ej{\ruby{今日}{きょう}は\ruby{昨日}{きのう}より\ruby{寒}{さむ}いです。}{Hoy hace más frío que ayer.}}

\begin{tcolorbox}[ejerciciobox, title={📝 Ejercicios Rápidos}]
\begin{enumerate}
  \item ¿Cómo dices "Está nevando"? (\ruby{降}{ふ}っています)
  \item Di cuál es tu estación favorita.
  \item Traduce: \ruby{冬}{ふゆ}はとても\ruby{寒}{さむ}いです。
\end{enumerate}
\vspace{0.4em}
\textbf{Respuestas:}
1) \ruby{雪}{ゆき}が\ruby{降}{ふ}っています。\quad
2) (personal: Ej. \ruby{秋}{あき}が\ruby{好}{す}きです)\quad
3) El invierno es muy frío.
\end{tcolorbox}
