\subsection*{⚙️ Gramática}

\patron{Acción en progreso}
  {V-て + います}
  {\ej{\ruby{雨}{あめ}が\ruby{降}{ふ}っています。}{Está lloviendo.}
  \ej{ご飯を\ruby{食}{た}べています。}{Estoy comiendo.}}

\patron{Estado resultado de una acción}
  {V-て + います (estados duraderos)}
  {\ej{\ruby{田中}{たなか}さんは\ruby{結}{けっ}\ruby{婚}{こん}しています。}{El Sr. Tanaka está casado.}
  \ej{\ruby{車}{くるま}を\ruby{持}{も}っています。}{Tengo un auto.}}
